/** * Biostatistics Learning Portal - Google Sheets backend * * Creates/uses two sheets: * Students - registration details (NEVER stores passwords) * Results - module quiz results * * IMPORTANT: * 1. Deploy as a Web App: Execute as Me, Who has access: Anyone. * 2. Replace API_KEY below with a long random value. * 3. Put the same API_KEY in both portal HTML files. * * The API key is an application key, not a student password. Because GitHub * Pages is public, this is NOT a high-security secret. Do not put passwords * or Google credentials in the HTML files. */ const API_KEY = 'GLUK_CLINICALMEDBIOSTAT_2026_StudentRecords_7K9mQ4xP'; const STUDENTS_SHEET = 'Students'; const RESULTS_SHEET = 'Results'; function setupPortalSheets() { const ss = SpreadsheetApp.getActiveSpreadsheet(); let students = ss.getSheetByName(STUDENTS_SHEET); if (!students) students = ss.insertSheet(STUDENTS_SHEET); if (students.getLastRow() === 0) { students.appendRow(['Registration No.', 'Student Name', 'Campus', 'Registered At', 'Last Seen', 'Password Hash']); } else if (students.getLastColumn() < 6) { students.getRange(1, 6).setValue('Password Hash'); } let results = ss.getSheetByName(RESULTS_SHEET); if (!results) results = ss.insertSheet(RESULTS_SHEET); if (results.getLastRow() === 0) { results.appendRow(['Registration No.', 'Student Name', 'Campus', 'Module', 'Score', 'Total', 'Percentage', 'Attempt', 'Submitted At']); } } function doGet(e) { const params = e && e.parameter ? e.parameter : {}; if (params.apiKey !== API_KEY) return json_({ ok: false, error: 'Unauthorized' }); if (params.action === 'getRecords') { return json_({ ok: true, data: getRecords_() }); } return json_({ ok: true, message: 'Biostatistics portal backend is running.' }); } function doPost(e) { try { const body = JSON.parse(e.postData.contents || '{}'); if (body.apiKey !== API_KEY) return json_({ ok: false, error: 'Unauthorized' }); setupPortalSheets(); if (body.action === 'register') { return json_({ ok: true, data: registerStudent_(body) }); } if (body.action === 'authenticate') { return json_({ ok: true, data: authenticateStudent_(body) }); } if (body.action === 'quizResult') { return json_({ ok: true, data: saveQuizResult_(body) }); } if (body.action === 'getRecords') { return json_({ ok: true, data: getRecords_() }); } return json_({ ok: false, error: 'Unknown action' }); } catch (err) { return json_({ ok: false, error: String(err && err.message ? err.message : err) }); } } function registerStudent_(body) { const reg = normalize_(body.registrationNumber); const name = clean_(body.name); const campus = clean_(body.campus); const password = String(body.password || ''); if (!reg || !name || !campus) throw new Error('Missing registration details.'); if (reg.length > 80 || name.length > 150 || campus.length > 100) throw new Error('Registration details are too long.'); if (password.length < 6 || password.length > 200) throw new Error('Invalid password.'); const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(STUDENTS_SHEET); const values = sheet.getDataRange().getValues(); const now = new Date(); const passwordHash = hashPassword_(password); for (let i = 1; i < values.length; i++) { const existingReg = normalize_(values[i][0]); if (existingReg === reg) { const createdAt = values[i][3] || now; sheet.getRange(i + 1, 2, 1, 5).setValues([[name, campus, createdAt, now, passwordHash]]); return { status: 'updated' }; } } sheet.appendRow([reg, name, campus, now, now, passwordHash]); return { status: 'created' }; } function authenticateStudent_(body) { const reg = normalize_(body.registrationNumber); const password = String(body.password || ''); if (!reg || !password) throw new Error('Missing login details.'); const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(STUDENTS_SHEET); const values = sheet.getDataRange().getValues(); const suppliedHash = hashPassword_(password); for (let i = 1; i < values.length; i++) { if (normalize_(values[i][0]) !== reg) continue; const storedHash = String(values[i][5] || ''); if (!storedHash || storedHash !== suppliedHash) return null; sheet.getRange(i + 1, 5).setValue(new Date()); return { registrationNumber: normalize_(values[i][0]), name: clean_(values[i][1]), campus: clean_(values[i][2]), createdAt: values[i][3] ? new Date(values[i][3]).toISOString() : '' }; } return null; } function hashPassword_(password) { const digest = Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, password, Utilities.Charset.UTF_8); return digest.map(function(byte) { const value = byte < 0 ? byte + 256 : byte; return ('0' + value.toString(16)).slice(-2); }).join(''); } function saveQuizResult_(body) { const reg = normalize_(body.registrationNumber); const name = clean_(body.name); const campus = clean_(body.campus); const moduleNumber = Number(body.moduleNumber); const score = Number(body.score); const total = Number(body.total); const percentage = Number(body.percentage); const attempt = Number(body.attempt); if (!reg || !name || !campus) throw new Error('Missing student identity.'); if (!Number.isInteger(moduleNumber) || moduleNumber < 1 || moduleNumber > 12) throw new Error('Invalid module.'); if (!Number.isFinite(score) || !Number.isFinite(total) || total <= 0 || score < 0 || score > total) throw new Error('Invalid score.'); if (!Number.isFinite(percentage) || percentage < 0 || percentage > 100) throw new Error('Invalid percentage.'); if (!Number.isInteger(attempt) || attempt < 1 || attempt > 2) throw new Error('Invalid attempt.'); const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(RESULTS_SHEET); const values = sheet.getDataRange().getValues(); const now = new Date(); // Keep the best score for each student/module, while retaining the latest // attempt number and timestamp when the score is equal or higher. for (let i = 1; i < values.length; i++) { const rowReg = normalize_(values[i][0]); const rowModule = Number(values[i][3]); if (rowReg === reg && rowModule === moduleNumber) { const oldPercentage = Number(values[i][6]); if (percentage >= oldPercentage) { sheet.getRange(i + 1, 1, 1, 9).setValues([[reg, name, campus, moduleNumber, score, total, percentage, attempt, now]]); } return { status: 'updated' }; } } sheet.appendRow([reg, name, campus, moduleNumber, score, total, percentage, attempt, now]); return { status: 'created' }; } function getRecords_() { const ss = SpreadsheetApp.getActiveSpreadsheet(); const students = ss.getSheetByName(STUDENTS_SHEET); const results = ss.getSheetByName(RESULTS_SHEET); const studentValues = students ? students.getDataRange().getValues() : []; const resultValues = results ? results.getDataRange().getValues() : []; const studentList = []; const resultMap = {}; for (let i = 1; i < studentValues.length; i++) { const reg = normalize_(studentValues[i][0]); if (!reg) continue; studentList.push({ registrationNumber: reg, name: clean_(studentValues[i][1]), campus: clean_(studentValues[i][2]), createdAt: studentValues[i][3] ? new Date(studentValues[i][3]).toISOString() : '', lastSeen: studentValues[i][4] ? new Date(studentValues[i][4]).toISOString() : '' }); } for (let i = 1; i < resultValues.length; i++) { const reg = normalize_(resultValues[i][0]); const moduleNumber = Number(resultValues[i][3]); if (!reg || !moduleNumber) continue; if (!resultMap[reg]) resultMap[reg] = {}; resultMap[reg]['module' + moduleNumber] = { score: Number(resultValues[i][4]), total: Number(resultValues[i][5]), percentage: Number(resultValues[i][6]), attempt: Number(resultValues[i][7]), submittedAt: resultValues[i][8] ? new Date(resultValues[i][8]).toISOString() : '' }; } return { students: studentList, results: resultMap }; } function normalize_(value) { return String(value == null ? '' : value).trim().toUpperCase(); } function clean_(value) { return String(value == null ? '' : value).trim(); } function json_(obj) { return ContentService .createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); }
